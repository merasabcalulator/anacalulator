            function processVideoUrl(url) {
                // Handle different video platforms
                if (url.includes('youtube.com/embed/')) {
                    // Add YouTube parameters for better embedding
                    const separator = url.includes('?') ? '&' : '?';
                    return url + separator + 'autoplay=1&rel=0&modestbranding=1';
                } else if (url.includes('youtu.be/')) {
                    // Convert youtu.be links to embed format
                    const videoId = url.split('youtu.be/')[1].split('?')[0];
                    return `https://www.youtube.com/embed/${videoId}?autoplay=1&rel=0&modestbranding=1`;
                } else if (url.includes('vimeo.com/')) {
                    // Handle Vimeo URLs
                    const videoId = url.split('vimeo.com/')[1].split('?')[0];
                    return `https://player.vimeo.com/video/${videoId}?autoplay=1`;
                }
                
                // Return original URL for other sources
                return url;
            }

                    // Try direct fetch first, then fallback
                    try {
                        const response = await fetch(apiUrl);
                        if (response.ok) {
                            const data = await response.json();
                            updateWeatherDisplay(data);
                            return;
                        }
                    } catch (e) {
                        console.log('Direct API failed, using fallback');
                    }
                    
                    // Fallback demo data
                    updateWeatherDisplay({
                        main: { temp: 39 },
                        weather: [{ main: 'Clear' }]
                    });
                } catch (error) {
                    console.log('Weather API error, using fallback');
                    updateWeatherDisplay({

                    });
                }

        async function handleAINewsGeneration(e) {
            e.preventDefault();
            
            const category = document.getElementById('newsCategory').value;
            const prompt = document.getElementById('newsPrompt').value;
            const style = document.getElementById('newsStyle').value;
            const generateBtn = document.getElementById('generateNewsBtn');
            const outputDiv = document.getElementById('generatedNews');
            const actionsDiv = document.querySelector('.news-actions');
            
            if (!category || !prompt) {
                alert('Please select a category and enter a news topic');
                return;
            }
            
            // Check if it's a search request
            const isSearchRequest = style === 'search' || prompt.toLowerCase().startsWith('search:');
            const searchQuery = prompt.toLowerCase().startsWith('search:') ? prompt.substring(7).trim() : prompt;
            
            // Show loading state
            generateBtn.disabled = true;
            generateBtn.innerHTML = `<i class="fas fa-spinner fa-spin"></i> ${isSearchRequest ? 'Searching...' : 'Generating...'}`;
            outputDiv.innerHTML = `<div class="placeholder-text">${isSearchRequest ? 'Searching latest news...' : 'Generating AI news content...'}</div>`;
            
            try {
                console.log('🚀 Starting AI news operation:', { category, prompt, style, isSearchRequest });
                
                let newsContent;
                if (isSearchRequest) {
                    newsContent = await searchLatestNews(searchQuery, category);
                } else {
                    newsContent = await generateNewsContent(prompt, category, style);
                }
                
                // Display generated content
                outputDiv.innerHTML = newsContent;
                actionsDiv.style.display = 'flex';
                
                // Add to AI news list
                addToAINewsList(newsContent, category, isSearchRequest ? 'search' : 'generated');
                
            } catch (error) {
                console.error('❌ AI News operation failed:', error);
                outputDiv.innerHTML = `<div class="placeholder-text" style="color: var(--danger);">Error: ${error.message}</div>`;
            } finally {
                // Reset button
                generateBtn.disabled = false;
                generateBtn.innerHTML = '<i class="fas fa-robot"></i> Generate News';
            }
        }
        
        async function searchLatestNews(query, category) {
            console.log('🔍 Starting news search:', { query, category });
            
            if (!window.ywConfig?.ai_config?.news_searcher) {
                throw new Error('API Error - News search configuration not found');
            }
            
            const config = window.ywConfig.ai_config.news_searcher;
            const systemPrompt = config.system_prompt({
                searchTopic: query,
                region: 'Kuwait',
                format: 'news article',
                audience: 'Kuwait residents and international readers'
            });
            
            const openai = window.aiSdk.openai.createOpenAI({
                baseURL: 'https://api.youware.com/public/v1/ai',
                apiKey: 'sk-YOUWARE'
            });
            
            try {
                const { text } = await window.aiSdk.ai.generateText({
                    model: openai(config.model),
                    messages: [
                        { role: 'system', content: systemPrompt },
                        { role: 'user', content: `Search and compile latest news about: ${query}. Category: ${category}. Please provide current, factual information with sources when possible. Focus on recent developments and provide a comprehensive news summary.` }
                    ],
                    temperature: config.temperature || 0.7,
                    maxTokens: config.maxTokens || 3000
                });
                
                console.log('✅ News search completed');
                return text;
            } catch (error) {
                console.error('❌ API Error - News search failed:', error);
                throw new Error(`API Error - News search failed: ${error.message}`);
            }
        }

        function addToAINewsList(content, category, type = 'generated') {
            const aiNewsList = document.getElementById('aiNewsList');
            const articlesList = aiNewsList.querySelector('.ai-news-list') || aiNewsList;
            
            // Extract title from content (first line or first sentence)
            const title = content.split('\n')[0].replace(/^#+\s*/, '') || content.substring(0, 60) + '...';
            const preview = content.substring(0, 150) + '...';
            const timestamp = new Date().toLocaleDateString();
            const articleId = 'ai-' + Date.now();
            const typeLabel = type === 'search' ? 'Searched' : 'Generated';
            const typeIcon = type === 'search' ? 'fas fa-search' : 'fas fa-robot';
            
            const articleElement = document.createElement('div');
            articleElement.className = 'ai-article-item';
            articleElement.innerHTML = `
                <h4><i class="${typeIcon}"></i> ${title}</h4>
                <p>${preview}</p>
                <small>${typeLabel} on: ${timestamp} | Category: ${category}</small>
                <div class="article-actions">
                    <button onclick="editAIArticle('${articleId}')">Edit</button>
                    <button onclick="publishAIArticle('${articleId}')">Publish</button>
                </div>
            `;
            
            // Store full content
            articleElement.dataset.fullContent = content;
            articleElement.dataset.articleId = articleId;
            articleElement.dataset.articleType = type;
            
            // Add to top of list
            if (articlesList.children.length > 1) {
                articlesList.insertBefore(articleElement, articlesList.children[1]);
            } else {
                articlesList.appendChild(articleElement);
            }
        }

        // Fetch news from Al Jazeera RSS feed
        async function fetchAlJazeeraNews() {
            try {
                const proxyUrl = 'https://api.allorigins.win/get?url=';
                const response = await fetch(proxyUrl + encodeURIComponent(newsSources['al-jazeera'].rssUrl));
                const data = await response.json();
                
                const parser = new DOMParser();
                const xmlDoc = parser.parseFromString(data.contents, 'text/xml');
                const items = xmlDoc.querySelectorAll('item');
                
                const articles = [];
                items.forEach((item, index) => {
                    if (index < 15) { // Limit to 15 articles
                        const title = item.querySelector('title')?.textContent || 'No title';
                        const description = item.querySelector('description')?.textContent || 'No description';
                        const link = item.querySelector('link')?.textContent || '#';
                        const pubDate = item.querySelector('pubDate')?.textContent || '';
                        
                        articles.push({
                            source: 'al-jazeera',
                            title: title,
                            description: description.replace(/<[^>]*>/g, ''), // Remove HTML tags
                            link: link,
                            publishedAt: new Date(pubDate),
                            image: 'https://www.aljazeera.com/images/logo_aje.png'
                        });
                    }
                });
                
                return articles;
            } catch (error) {
                console.error('Error fetching Al Jazeera news:', error);
                return [];
            }
        }

                    <div class="news-content">
                        <span class="news-source">${newsSources[article.source].name}</span>
                        <h3 class="news-title">${article.title}</h3>
                        <p class="news-description">${article.description.substring(0, 150)}...</p>
                        <div class="news-meta">
                            <span class="news-date">${formatNewsDate(article.publishedAt)}</span>
                            <a href="${article.link}" target="_blank" class="news-link">Read More</a>
                        </div>
                    </div>
                </article>
            `).join('');
        }

        // Format news date for display
        function formatNewsDate(date) {
            const now = new Date();
            const diffMs = now - new Date(date);
            const diffMins = Math.floor(diffMs / 60000);
            const diffHours = Math.floor(diffMs / 3600000);
            const diffDays = Math.floor(diffMs / 86400000);

            if (diffMins < 1) return 'Just now';
            if (diffMins < 60) return `${diffMins}m ago`;
            if (diffHours < 24) return `${diffHours}h ago`;
            if (diffDays < 7) return `${diffDays}d ago`;
            
            return new Date(date).toLocaleDateString();
        }

        // Toggle auto-refresh
        function toggleAutoRefresh() {
            const toggle = document.getElementById('autoRefreshToggle');
            
            if (toggle.checked) {
                // Start auto-refresh every 30 seconds
                newsAutoRefreshInterval = setInterval(fetchAllNews, 30000);
                console.log('Auto-refresh enabled');
            } else {
                // Stop auto-refresh
                if (newsAutoRefreshInterval) {
                    clearInterval(newsAutoRefreshInterval);
                    newsAutoRefreshInterval = null;
                }
                console.log('Auto-refresh disabled');
            }

        // Update the showView function to handle news
        function showView(viewName) {
            // Hide all views
            const views = ['home', 'news', 'games', 'leaderboard', 'about', 'contact'];
            views.forEach(view => {
                const element = document.getElementById(view + 'View');
                if (element) {
                    element.style.display = 'none';
                }
            });

            // Handle main content sections
            const mainSections = document.querySelectorAll('.main-content > .container, .view-section');
            mainSections.forEach(section => {
                if (viewName === 'home') {
                    section.style.display = section.id === viewName + 'View' ? 'block' : 
                                          section.classList.contains('view-section') ? 'none' : 'block';
                } else {
                    section.style.display = section.id === viewName + 'View' ? 'block' : 'none';
                }
            });

            // Show specific view
            if (viewName === 'news') {
                document.getElementById('newsView').style.display = 'block';
                fetchAllNews(); // Load news when showing news view
            }

            currentView = viewName;
        }
